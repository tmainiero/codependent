\section{Categories}\label{sec:cats}

\begin{definition}\label{def:cat}
  A \textbf{category} $\Cat*$ consists of objects and morphisms,
  with a $\Hom*{X}{Y}$ set for each pair of objects $X, Y$.
\end{definition}

\begin{proposition}\label{prop:comp}
  Composition in $\Cat$ is associative. See \cref{def:cat}.
\end{proposition}

\begin{theorem}\label{thm:main}
  Every small category admits a faithful functor to $\Cat$.
  This follows from \cref{prop:comp} and \cref{def:cat}.
\end{theorem}
