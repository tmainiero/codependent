\section{Groups and subgroups}

\begin{definition}\label{def:group}
  A \emph{group} is a set $G$ with a binary operation satisfying
  associativity, identity, and inverse axioms.
\end{definition}

\begin{theorem}\label{thm:lagrange}
  If $H \le G$ and $G$ is finite, then $|H|$ divides $|G|$.
\end{theorem}

\begin{proof}
  The left cosets $gH$ partition $G$ into sets of equal size $|H|$.
\end{proof}

\section{Normal subgroups}

\begin{definition}\label{def:normal}
  A subgroup $N \le G$ is \emph{normal}, written $N \trianglelefteq G$,
  if $gNg^{-1} = N$ for all $g \in G$.
\end{definition}

\begin{theorem}\label{thm:quotient}
  If $N \trianglelefteq G$ (\cref{def:normal}), then $G/N$ is a group.
\end{theorem}

\begin{proof}
  Well-definedness follows from normality (\cref{def:normal}).
\end{proof}
